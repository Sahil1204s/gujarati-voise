    <textarea id="textInput" placeholder="અહીં ગુજરાતીમાં લખાણ લખો અથવા પેસ્ટ કરો..."></textarea>

    <div class="controls">
      <label for="rate">સ્પીડ (Speed): <span id="rateValue">1</span>x</label>
      <input type="range" id="rate" min="0.5" max="1.5" value="1" step="0.1">

      <label for="pitch">અવાજની પિચ (Pitch): <span id="pitchValue">1</span></label>
      <input type="range" id="pitch" min="0.5" max="1.5" value="1" step="0.1">
    </div>

    <div class="btn-group">
      <button class="play-btn" onclick="speakText()">અવાજ સાંભળો</button>
      <button class="stop-btn" onclick="stopText()">બંધ કરો</button>
    </div>

    <p class="status" id="statusMsg">તૈયાર છે</p>
  </div>

  <script>
    const synth = window.speechSynthesis;
    const textInput = document.getElementById('textInput');
    const rateInput = document.getElementById('rate');
    const pitchInput = document.getElementById('pitch');
    const rateValue = document.getElementById('rateValue');
    const pitchValue = document.getElementById('pitchValue');
    const statusMsg = document.getElementById('statusMsg');

    rateInput.addEventListener('input', () => rateValue.textContent = rateInput.value);
    pitchInput.addEventListener('input', () => pitchValue.textContent = pitchInput.value);

    function speakText() {
      if (synth.speaking) {
        synth.cancel();
      }

      const text = textInput.value.trim();
      if (!text) {
        alert('કૃપા કરીને કંઈક લખાણ દાખલ કરો!');
        return;
      }

      const utterance = new SpeechSynthesisUtterance(text);
      utterance.lang = 'gu-IN'; // ગુજરાતી ભાષા માટેનો કોડ
      utterance.rate = parseFloat(rateInput.value);
      utterance.pitch = parseFloat(pitchInput.value);

      // સિસ્ટમમાં ઉપલબ્ધ ગુજરાતી અવાજ પસંદ કરવો
      const voices = synth.getVoices();
      const gujaratiVoice = voices.find(voice => voice.lang === 'gu-IN' || voice.lang.includes('gu'));
      if (gujaratiVoice) {
        utterance.voice = gujaratiVoice;
      }

      utterance.onstart = () => { statusMsg.textContent = 'અવાજ ચાલી રહ્યો છે...'; };
      utterance.onend = () => { statusMsg.textContent = 'પૂર્ણ થયું'; };
      utterance.onerror = () => { statusMsg.textContent = 'કંઈક ભૂલ થઈ!'; };

      synth.speak(utterance);
    }

    function stopText() {
      if (synth.speaking) {
        synth.cancel();
        statusMsg.textContent = 'અટકાવી દીધું';
      }
    }

    if (speechSynthesis.onvoiceschanged !== undefined) {
      speechSynthesis.onvoiceschanged = () => synth.getVoices();
    }
  </script>
